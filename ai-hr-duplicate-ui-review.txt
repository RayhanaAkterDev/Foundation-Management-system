// --------------------------------

export const fetchHelpRequest = async (helpRequestId) => {
  return apiRequest(`/admin/help-requests/${helpRequestId}`);
};

// --------------------------------
// Help Requests - Update
// --------------------------------

export const updateHelpRequest = async (helpRequestId, requestData) => {
  if (!helpRequestId) {
    throw new Error("Help request ID is required.");
  }

  return apiRequest(`/admin/help-requests/${helpRequestId}`, {
    method: "PUT",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      title: requestData.title,
      description: requestData.description,
      category: requestData.category,
      urgency: requestData.urgency,
      district: requestData.district,
      address: requestData.address ?? null,
    }),
  });
};

// --------------------------------
// Help Requests - Verify / Reject
// --------------------------------

export const updateHelpRequestVerification = async (
  helpRequestId,
  status,
  verificationNote = null,
) => {
  return apiRequest(`/admin/help-requests/${helpRequestId}/verification`, {
    method: "PATCH",
    body: JSON.stringify({
      status,
      verification_note: verificationNote,
    }),
  });
};

// --------------------------------
// Help Requests - Urgency
// --------------------------------

export const updateHelpRequestUrgency = async (helpRequestId, urgency) => {
  return apiRequest(`/admin/help-requests/${helpRequestId}/urgency`, {
    method: "PATCH",
    body: JSON.stringify({
      urgency,
    }),
  });
};

// --------------------------------
// Help Requests - Assignment

export const assignHelpRequest = async (helpRequestId, assignmentData) => {
  return apiRequest(`/admin/help-requests/${helpRequestId}/assignment`, {
    method: "PATCH",

                            assignment?.organization?.name ||

                            'The assigned organization';

                            request?.title ||
                            assignment?.help_request?.title ||

function show(value, fallback = 'Not provided') {
    if (value === null || value === undefined) {
        return fallback;
    }

    const result = String(value).trim();
    return result || fallback;
}

function titleCase(value) {
    return show(value, 'Not specified')
        .replace(/[_-]+/g, ' ')
        .replace(/\b\w/g, (character) => character.toUpperCase());
}

function dateText(value, withTime = false) {
    if (!value) return 'Not set';

    const parsed = new Date(value);

    if (Number.isNaN(parsed.getTime())) {
        return String(value);
    }

    const options = {
        day: 'numeric',
        month: 'short',
        year: 'numeric',
    };

    if (withTime) {
        options.hour = 'numeric';
        options.minute = '2-digit';
    }

    return parsed.toLocaleString('en-GB', options);
}

function getAssignments(request) {
    if (Array.isArray(request.assignments)) {
        return request.assignments;
    }

    if (Array.isArray(request.help_request_assignments)) {
        return request.help_request_assignments;
    }

    if (Array.isArray(request.helpRequestAssignments)) {
        return request.helpRequestAssignments;
    }

    return request.assignment ? [request.assignment] : [];
}
